75typedef std::map<int, std::set<std::pair<int, int> > > BAOutliers;
76
78inline bool baObservationRejected(const BAOutliers & outliers, int wordId, int poseId, int cameraIndex)
79{
80 BAOutliers::const_iterator iter = outliers.find(wordId);
81 return iter != outliers.end() && iter->second.count(std::make_pair(poseId, cameraIndex)) != 0;
82}
83
89inline bool baPoseRejected(const BAOutliers & outliers, const std::multimap<int, FeatureBA> & references, int wordId, int poseId)
90{
91 bool observed = false;
92 for(std::multimap<int, FeatureBA>::const_iterator iter=references.lower_bound(poseId); iter!=references.end() && iter->first == poseId; ++iter)
93 {
94 observed = true;
95 if(!baObservationRejected(outliers, wordId, poseId, iter->second.cameraIndex))
96 {
97 return false;
98 }
99 }
100 return observed;
101}
