105 {

461inline T uMeanSquaredError(const T * x, unsigned int sizeX, const T * y, unsigned int sizeY)
462{
463 T sum = 0;
464 if(x && y && sizeX == sizeY)
465 {
466 for(unsigned int i=0; i<sizeX; ++i)
467 {
468 T diff = x[i]-y[i];

470 }
471 return sum/(T)sizeX;
472 }
473 return (T)-1;

652template<class T>
653inline std::list<unsigned int> uLocalMaxima(const T * v, unsigned int size)
654{
655 std::list<unsigned int> maxima;
656 if(size)
657 {
658 for(unsigned int i=0; i<size; ++i)
659 {
660 if(i == 0)
661 {
662 // first item
663 if((i+1 < size && v[i] > v[i+1]) ||
664 i+1 >= size)
665 {
666 maxima.push_back(i);
667 }
668 }
669 else if(i == size - 1)
670 {
671 //last item
672 if((i >= 1 && v[i] > v[i-1]) ||
673 i == 0)
674 {
675 maxima.push_back(i);
676 }
677 }
678 else
679 {
680 //all others, check previous and next
681 if(v[i] > v[i-1] && v[i] > v[i+1])
682 {
683 maxima.push_back(i);
684 }
685 }
686 }
687 }
688 return maxima;
689}

715template<class T>
716inline std::vector<T> uXMatch(const T * vA, const T * vB, unsigned int sizeA, unsigned int sizeB, UXMatchMethod method)
717{
718 if(!vA || !vB || sizeA == 0 || sizeB == 0)
719 {
720 return std::vector<T>();
721 }
722
723 std::vector<T> result(sizeA + sizeB - 1);
724
725 T meanA = 0;
726 T meanB = 0;
727 if(method > UXCorrCoeff)
728 {
729 meanA = uMean(vA, sizeA);
730 meanB = uMean(vB, sizeB);
731 }
732
733 T den = 1;
734 if(method == UXCorrCoeff || method == UXCovCoeff)
735 {
736 den = std::sqrt(uSumSquared(vA, sizeA, meanA) * uSumSquared(vB, sizeB, meanB));
737 }
738 else if(method == UXCorrBiased || method == UXCovBiased)
739 {
740 den = (T)std::max(sizeA, sizeB);
741 }
742
743 if(sizeA == sizeB)
744 {
745 T resultA;
746 T resultB;
747
748 int posA;
749 int posB;
750 unsigned int j;
751
752 // Optimization, filling two results at once
753 for(unsigned int i=0; i<sizeA; ++i)
754 {
755 if(method == UXCorrUnbiased || method == UXCovUnbiased)
756 {
757 den = 0;
758 }
759
760 posA = sizeA - i - 1;
761 posB = sizeB - i - 1;
762 resultA = 0;
763 resultB = 0;
764 for(j=0; (j + posB) < sizeB && (j + posA) < sizeA; ++j)
765 {
766 resultA += (vA[j] - meanA) * (vB[j + posB] - meanB);
767 resultB += (vA[j + posA] - meanA) * (vB[j] - meanB);
768 if(method == UXCorrUnbiased || method == UXCovUnbiased)
769 {
770 ++den;
771 }
772 }
773
774 result[i] = resultA / den;
775 result[result.size()-1 -i] = resultB / den;
776 }
777 }
778 else
779 {
780 for(unsigned int i=0; i<result.size(); ++i)
781 {
782 if(method == UXCorrUnbiased || method == UXCovUnbiased)
783 {
784 den = 0;
785 }
786
787 int posB = sizeB - i - 1;
788 T r = 0;
789 if(posB >= 0)
790 {
791 for(unsigned int j=0; (j + posB) < sizeB && j < sizeA; ++j)
792 {
793 r += (vA[j] - meanA) * (vB[j + posB] - meanB);
794 if(method == UXCorrUnbiased || method == UXCovUnbiased)
795 {
796 ++den;
797 }
798 }
799 }
800 else
801 {
802 int posA = posB*-1;
803 for(unsigned int i=0; (i+posA) < sizeA && i < sizeB; ++i)
804 {
805 r += (vA[i+posA] - meanA) * (vB[i] - meanB);
806 if(method == UXCorrUnbiased || method == UXCovUnbiased)
807 {
808 ++den;
809 }
810 }
811 }
812
813 result[i] = r / den;
814 }
815 }
816
817 return result;
818}

845{
846 T result = 0;
847 if(!vA || !vB || sizeA == 0 || sizeB == 0)
848 {
849 return result;
850 }
851
852 T meanA = 0;
853 T meanB = 0;
854 if(method > UXCorrCoeff)
855 {
856 meanA = uMean(vA, sizeA);
857 meanB = uMean(vB, sizeB);
858 }
859 unsigned int size = sizeA + sizeB - 1;
860
861 T den = 1;
862 if(method == UXCorrCoeff || method == UXCovCoeff)
863 {
864 den = std::sqrt(uSumSquared(vA, sizeA, meanA) * uSumSquared(vB, sizeB, meanB));
865 }
866 else if(method == UXCorrBiased || method == UXCovBiased)
867 {
868 den = (T)std::max(sizeA, sizeB);
869 }
870 else if(method == UXCorrUnbiased || method == UXCovUnbiased)
871 {
872 den = 0;
873 }
874
875 if(index < size)
876 {
877 int posB = sizeB - index - 1;
878 unsigned int i;
879 if(posB >= 0)
880 {
881 for(i=0; (i + posB) < sizeB && i < sizeA; ++i)
882 {
883 result += (vA[i] - meanA) * (vB[i + posB] - meanB);
884 if(method == UXCorrUnbiased || method == UXCovUnbiased)
885 {
886 ++den;
887 }
888 }
889 }

892 int posA = posB*-1;
893 for(i=0; (i+posA) < sizeA && i < sizeB; ++i)

895 result += (vA[i+posA] - meanA) * (vB[i] - meanB);
896 if(method == UXCorrUnbiased || method == UXCovUnbiased)

898 ++den;
899 }
900 }
901 }

927inline std::vector<float> uHamming(unsigned int L)
928{
929 std::vector<float> w(L);
930 unsigned int N = L-1;
931 float pi = 3.14159265f;
932 for(unsigned int n=0; n<N; ++n)
933 {
934 w[n] = 0.54f-0.46f*std::cos(2.0f*pi*float(n)/float(N));
935 }
936 return w;
937}
