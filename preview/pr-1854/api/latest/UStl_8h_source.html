53template<class K, class V>
54inline std::list<K> uUniqueKeys(const std::multimap<K, V> & mm)
55{
56 std::list<K> l;
57 typename std::list<K>::reverse_iterator lastValue;
58 for(typename std::multimap<K, V>::const_iterator iter = mm.begin(); iter!=mm.end(); ++iter)
59 {
60 if(iter == mm.begin() || (iter != mm.begin() && *lastValue != iter->first))
61 {
62 l.push_back(iter->first);
63 lastValue = l.rbegin();
64 }
65 }
66 return l;
67}

510template<class K, class V>
511inline std::map<K, V> uMultimapToMapUnique(const std::multimap<K, V> & m)
512{
513 std::map<K, V> mapOut;
514 std::list<K> uniqueKeys = uUniqueKeys(m);
515 for(typename std::list<K>::const_iterator iter = uniqueKeys.begin(); iter!=uniqueKeys.end(); ++iter)
516 {
517 if(m.count(*iter) == 1)
518 {
519 typename std::multimap<K, V>::const_iterator jter=m.find(*iter);
520 mapOut.insert(mapOut.end(), std::pair<K,V>(jter->first, jter->second));
521 }
522 }
523 return mapOut;
524}

678inline std::list<std::string> uSplitNumChar(const std::string & str)
679{
680 std::list<std::string> list;
681 std::string buf;
682 bool num = false;
683 for(unsigned int i=0; i<str.size(); ++i)
684 {
685 if(uIsDigit(str[i]))
686 {
687 if(!num && buf.size())
688 {
689 list.push_back(buf);
690 buf.clear();
691 }
692 buf += str[i];
693 num = true;
694 }
695 else
696 {
697 if(num)
698 {
699 list.push_back(buf);
700 buf.clear();
701 }
702 buf += str[i];
703 num = false;
704 }
705 }
706 if(buf.size())
707 {
708 list.push_back(buf);
709 }
710 return list;
711}

725inline int uStrNumCmp(const std::string & a, const std::string & b)
726{
727 std::vector<std::string> listA;
728 std::vector<std::string> listB;
729
730 listA = uListToVector(uSplitNumChar(a));
731 listB = uListToVector(uSplitNumChar(b));
732
733 unsigned int i;
734 int result = 0;
735 for(i=0; i<listA.size() && i<listB.size(); ++i)
736 {
737 if(uIsDigit(listA[i].at(0)) && uIsDigit(listB[i].at(0)))
738 {
739 //padding if zeros at the beginning
740 if(listA[i].at(0) == '0' && listB[i].size() < listA[i].size())
741 {
742 while(listB[i].size() < listA[i].size())
743 {
744 listB[i] += '0';
745 }
746 }
747 else if(listB[i].at(0) == '0' && listA[i].size() < listB[i].size())
748 {
749 while(listA[i].size() < listB[i].size())
750 {
751 listA[i] += '0';
752 }
753 }
754
755 if(listB[i].size() < listA[i].size())
756 {
757 result = 1;
758 }
759 else if(listB[i].size() > listA[i].size())
760 {
761 result = -1;
762 }
763 else
764 {
765 result = listA[i].compare(listB[i]);
766 }
767 }
768 else if(uIsDigit(listA[i].at(0)))
769 {
770 result = -1;
771 }
772 else if(uIsDigit(listB[i].at(0)))
773 {
774 result = 1;
775 }
776 else
777 {
778 result = listA[i].compare(listB[i]);
779 }
780
781 if(result != 0)
782 {
783 break;
784 }
785 }
786
787 return result;
788}

798inline int uCompareVersion(const std::string & version, int major, int minor=-1, int patch=-1)
799{
800 std::vector<std::string> v = uListToVector(uSplit(version, '.'));
801 if(v.size() == 3)
802 {
803 int vMajor = atoi(v[0].c_str());
804 int vMinor = atoi(v[1].c_str());
805 int vPatch = atoi(v[2].c_str());
806 if(vMajor > major ||
807 (vMajor == major && minor!=-1 && vMinor > minor) ||
808 (vMajor == major && minor!=-1 && vMinor == minor && patch!=-1 && vPatch > patch))
809 {
810 return 1;
811 }
812 else if(vMajor == major && (minor == -1 || (vMinor == minor && (patch == -1 || vPatch == patch))))
813 {
814 return 0;
815 }
816 }
817 return -1;
818}
