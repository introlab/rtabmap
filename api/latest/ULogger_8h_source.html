389 /*
390 * Write a message to logger: use UDEBUG(), UINFO(), UWARN(), UERROR() or UFATAL() instead.
391 * @param level the log level of this message
392 * @param file the file path
393 * @param line the line in the file
394 * @param function the function name in which the message is logged
395 * @param msg the message to write
396 * @param ... the variable arguments

399 const char * file,
400 int line,
401 const char *function,
402 const char* msg,
403 ...)
404#if defined(__GNUC__) || defined(__clang__)
405 // Let the compiler type-check every UDEBUG/UINFO/UWARN/UERROR
406 // format string against its arguments. Without this, a mismatch is
407 // invisible until the line actually executes -- and since
408 // ULogger::write() returns early when the message is below the
409 // current log level, a bad format can sit latent for years and then
410 // segfault only for users running at a lower level. `msg` is the
411 // 5th parameter (no implicit `this`: the function is static) and
412 // the varargs start at 6.
413 __attribute__((format(printf, 5, 6)))
414#endif
