170 struct Sample
171 {
172 Eigen::Quaterniond orientation;
173 // Kept with gravity, which is removed when used: the estimate can change
174 Eigen::Vector3d specificForce;
175 bool hasAcceleration;
176 };
177 // Acceleration of a sample, gravity removed (null without acceleration)
178 Eigen::Vector3d accelerationOf(const Sample & sample) const;

183 void updateIntegration() const;

185 // The acceleration integrated from the last pose up to a sample's stamp, in the
186 // odometry frame: predicting a stamp then only integrates from the sample before it.
187 struct Integrated
188 {
189 Eigen::Vector3d acceleration; // at that stamp
190 Eigen::Vector3d velocity; // change since the last pose
191 Eigen::Vector3d position; // change since the last pose (without its velocity)
192 };
193
194private:
195 double maxPoseInterval_;
196 double velocityWindow_;
197 double gravity_;
198 bool gravityEstimated_;
199 std::map<double, Sample> samples_;
200
201 // Gravity estimation: the samples of the current window, and the gravity measured over
202 // the last quiet windows
203 struct StillSample
204 {
205 double stamp;
206 Eigen::Vector3d specificForce; // in the IMU frame
207 double angularVelocity; // norm
208 };
209 std::vector<StillSample> stillWindow_;
210 std::vector<double> gravityWindows_;
211 size_t gravityWindowsCount_;
212 // Recent odometry poses, to estimate the velocity from (see velocityWindow)
213 std::map<double, rtabmap::Transform> poses_;
214
215 // The last odometry pose and the state predictions start from.
216 double poseStamp_;
217 rtabmap::Transform pose_;
218 Eigen::Vector3d velocity_;
219 // Rotation from the IMU's world frame to the odometry frame, at the last pose: the two
220 // are both gravity aligned, but their yaw differ.
221 Eigen::Quaterniond worldToOdom_;
222
223 // Built lazily by predict(), from the last pose to the newest sample; cleared when the
224 // pose or the samples it was built from change.
225 mutable std::map<double, Integrated> integrated_;
226 // Whether the acceleration at the last pose's stamp (the first entry) is final: it is
227 // held constant from the newest sample until a sample after that stamp is received.
228 mutable bool integratedStartIsFinal_;
229};
